from datetime import datetime, timedelta, timezone

from jose import jwt
from pwdlib import PasswordHash
from pwdlib.hashers.bcrypt import BcryptHasher

from app.core.config import settings

pwd_hash = PasswordHash((BcryptHasher(),))

def _truncate_password(password: str) -> str:
    # bcrypt only supports passwords up to 72 bytes.
    return password.encode("utf-8")[:72].decode("utf-8", errors="ignore")


def hash_password(password: str) -> str:
    return pwd_hash.hash(_truncate_password(password))


def verify_password(plain_password: str, hashed_password: str) -> bool:
    return pwd_hash.verify(
        _truncate_password(plain_password),
        hashed_password,
    )


def create_access_token(data: dict) -> str:
    to_encode = data.copy()

    expire = datetime.now(timezone.utc) + timedelta(
        minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES
    )

    to_encode.update({"exp": expire})

    return jwt.encode(
        to_encode,
        settings.SECRET_KEY,
        algorithm="HS256",
    )


def decode_access_token(token: str) -> dict | None:
    try:
        payload = jwt.decode(
            token,
            settings.SECRET_KEY,
            algorithms=["HS256"],
        )
        return payload
    except jwt.JWTError:
        return None

        status_code=status.HTTP_401_UNAUTHORIZED,

        headers={"WWW-Authenticate": "Bearer"},
    )

    payload = decode_access_token(token)
    if payload is None:
        raise credentials_exception

    # Usually stored under "sub" (either user email or user ID)
    user_id_or_email = payload.get("sub")
    if user_id_or_email is None:
        raise credentials_exception

    # Query the user from PostgreSQL
    if str(user_id_or_email).isdigit():
        stmt = select(User).where(User.id == int(user_id_or_email))
    else:
        stmt = select(User).where(User.email == str(user_id_or_email))

    result = await db.execute(stmt)
    user = result.scalar_one_or_none()

    if user is None:
        raise credentials_exception

    return user

@router.put("/{org_id}", response_model=BrandingOut)
async def update_branding(
    org_id: int,
    color: str = Form(...),
    assistant_name: str | None = Form(None),
    header_tagline: str | None = Form(None),
    font_family: str | None = Form(None),
    mascot: UploadFile | None = File(None),
    remove_mascot: bool = Form(False),
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    await require_branding_editor(db, org_id, current_user)

    if not HEX_COLOR.match(color):
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail="Color must look like #7A1F2B.",
        )

    normalized_name = assistant_name.strip() if assistant_name is not None else None
    if normalized_name is not None and not normalized_name:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail="Assistant name cannot be empty.",
        )
    if normalized_name is not None and len(normalized_name) > MAX_ASSISTANT_NAME_LENGTH:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail=f"Assistant name must be {MAX_ASSISTANT_NAME_LENGTH} characters or fewer.",
        )

    normalized_tagline = (
        header_tagline.strip() if header_tagline is not None else None
    )
    if normalized_tagline is not None and not normalized_tagline:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail="Header tagline cannot be empty.",
        )
    if (
        normalized_tagline is not None
        and len(normalized_tagline) > MAX_HEADER_TAGLINE_LENGTH
    ):
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail=f"Header tagline must be {MAX_HEADER_TAGLINE_LENGTH} characters or fewer.",
        )

    normalized_font_family = (
        font_family.strip().lower() if font_family is not None else None
    )
    if (
        normalized_font_family is not None
        and normalized_font_family not in SUPPORTED_FONT_FAMILIES
    ):
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
            detail="Choose a supported chat font.",
        )

    branding = await db.get(OrgBranding, org_id)
    if branding is None:
        branding = OrgBranding(org_id=org_id, color=color.upper())
        db.add(branding)
    branding.color = color.upper()

    chat_settings = await db.get(OrgChatSettings, org_id)
    if normalized_name is not None:
        if chat_settings is None:
            chat_settings = OrgChatSettings(
                org_id=org_id, assistant_name=normalized_name
            )
            db.add(chat_settings)
        else:
            chat_settings.assistant_name = normalized_name

    chat_header = await db.get(OrgChatHeader, org_id)
    if normalized_tagline is not None:
        if chat_header is None:
            chat_header = OrgChatHeader(
                org_id=org_id, header_tagline=normalized_tagline
            )
            db.add(chat_header)
        else:
            chat_header.header_tagline = normalized_tagline

    chat_font = await db.get(OrgChatFont, org_id)
    if normalized_font_family is not None:
        if chat_font is None:
            chat_font = OrgChatFont(
                org_id=org_id, font_family=normalized_font_family
            )
            db.add(chat_font)
        else:
            chat_font.font_family = normalized_font_family

    if mascot is not None and mascot.filename:
        data = await mascot.read(MAX_MASCOT_BYTES + 1)
        if len(data) > MAX_MASCOT_BYTES:
            raise HTTPException(
                status_code=status.HTTP_413_REQUEST_ENTITY_TOO_LARGE,
                detail="The mascot image must be 512 KB or smaller.",
            )
        # Check the file's real signature, not just the name or content type
        if not data.startswith(PNG_MAGIC):
            raise HTTPException(
                status_code=status.HTTP_415_UNSUPPORTED_MEDIA_TYPE,
                detail="The mascot must be a PNG image.",
            )
        branding.mascot_png = data
        branding.mascot_updated_at = datetime.utcnow()
    elif remove_mascot:
        branding.mascot_png = None
        branding.mascot_updated_at = None

    # Build the response before commit(): commit expires the object's attributes
    result = to_out(
        org_id,
        branding.color,
        branding.mascot_updated_at,
        normalized_name
        or (chat_settings.assistant_name if chat_settings else DEFAULT_ASSISTANT_NAME),
        normalized_tagline
        or (chat_header.header_tagline if chat_header else DEFAULT_HEADER_TAGLINE),
        normalized_font_family
        or (chat_font.font_family if chat_font else DEFAULT_FONT_FAMILY),
    )
    await db.commit()
    return result

async def _extract_uploaded_file(file: UploadFile) -> tuple[str, str]:
    filename = " ".join(
        Path((file.filename or "").replace("\\", "/")).name.split()
    )
    extension = Path(filename).suffix.lower()
    if extension not in {".pdf", ".doc", ".docx", ".xls", ".xlsx"}:
        raise HTTPException(
            status_code=status.HTTP_415_UNSUPPORTED_MEDIA_TYPE,
            detail="Only PDF, DOC, DOCX, XLS, and XLSX files are supported.",
        )
    if not filename or len(filename) > 255:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="The document filename must be between 1 and 255 characters.",
        )

    try:
        content = await file.read(MAX_DOCUMENT_UPLOAD_BYTES + 1)
    finally:
        await file.close()
    if len(content) > MAX_DOCUMENT_UPLOAD_BYTES:
        raise HTTPException(
            status_code=status.HTTP_413_REQUEST_ENTITY_TOO_LARGE,
            detail="Documents must be no larger than 25 MB.",
        )
    if not content:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="The uploaded document is empty.",
        )

    try:
        text = await asyncio.to_thread(extract_document_text, filename, content)
    except DocumentExtractionError as exc:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail=str(exc),
        ) from exc
    except DocumentParserUnavailableError as exc:
        raise HTTPException(
            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
            detail=str(exc),
        ) from exc
    return filename, text


def _document_markdown(filename: str, category: str, extracted_text: str) -> str:
    escaped_filename = filename.replace("\r", " ").replace("\n", " ")
    escaped_filename = escaped_filename.replace("#", "\\#")
    return f"# {escaped_filename}\n\nCategory: {category}\n\n{extracted_text}"

@router.get("/documents")
async def list_documents(
    org_id: int = Query(...),
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    membership = await db.execute(
        select(OrgMember).where(
            OrgMember.org_id == org_id,
            OrgMember.user_id == current_user.id,
        )
    )
    if membership.scalar_one_or_none() is None:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="You do not have access to this organization.",
        )

    pages_result = await db.execute(
        select(ScrapedPage, Website)
        .join(Website, ScrapedPage.web_id == Website.id)
        .where(
            Website.org_id == org_id,
            Website.url.startswith(DOCUMENT_URL_SCHEME),
        )
        .order_by(ScrapedPage.id.desc())
    )
    pages = pages_result.all()
    if not pages:
        return []

    page_ids = [page.id for page, _website in pages]
    sync_col = get_active_sync_column()
    count_rows = await db.execute(
        select(
            ScrapedPage.id,
            func.count(func.distinct(Chunk.id)),
            func.count(func.distinct(GeneratedQuestion.id)),
            func.count(func.distinct(GeneratedQuestion.id)).filter(sync_col.is_(True)),
        )
        .select_from(ScrapedPage)
        .outerjoin(Chunk, Chunk.page_id == ScrapedPage.id)
        .outerjoin(GeneratedQuestion, GeneratedQuestion.chunk_id == Chunk.id)
        .where(ScrapedPage.id.in_(page_ids))
        .group_by(ScrapedPage.id)
    )
    counts_by_id = {
        page_id: (chunks or 0, questions or 0, synced or 0)
        for page_id, chunks, questions, synced in count_rows.all()
    }

    documents = []
    for page, website in pages:
        chunks, questions, synced = counts_by_id.get(page.id, (0, 0, 0))
        category = ""
        for line in (page.markdown_content or "").splitlines():
            if line.startswith("Category: "):
                category = line.removeprefix("Category: ").strip()
                break
        documents.append(
            {
                "document_id": page.id,
                "filename": _document_filename(page),
                "category": category,
                "status": _document_status(page, website, questions, synced),
                "error_message": website.error_message,
                "chunks": chunks,
                "questions": questions,
                "synced_questions": synced,
            }
        )
    return documents

    await db.commit()

def _collect_titles(result: PipelineQueryResponse) -> tuple[list[str], list[str]]:
    chunk_titles: dict[str, None] = {}  # dict keeps order and removes duplicates
    questions: dict[str, None] = {}

    for ctx in result.contexts or []:
        if ctx.title:
            chunk_titles[ctx.title] = None
        for mq in ctx.matched_questions or []:
            if mq.question:
                questions[mq.question] = None

    return list(chunk_titles), list(questions)


@router.post("/{org_id}", response_model=PipelineQueryResponse)
async def execute_query(
    org_id: int,
    body: QueryBodyRequest,
    db: AsyncSession = Depends(get_db),
    pipeline: QueryPipeline = Depends(get_query_pipeline),
):
    org = await db.get(Org, org_id)
    if not org:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"Organization {org_id} not found.",
        )

    # Earlier, unexpired turns of this chat session (empty for the first message)
    history = await load_history(db, org_id, body.session_id)
    history = [
        turn.model_copy(
            update={"content": redact_personal_information(turn.content)}
        )
        for turn in history
    ]
    sanitized_query = redact_personal_information(body.query)

    pipeline_req = PipelineQueryRequest(
        org_id=org_id,
        query=sanitized_query,
        session_id=body.session_id,
        history=history,
    )

    started = time.perf_counter()
    result = await pipeline.execute(pipeline_req)  # raises -> nothing below runs
    latency_ms = int((time.perf_counter() - started) * 1000)

    titles, questions = _collect_titles(result)

    try:
        log = ResponseLog(
            org_id=org_id,
            latency_ms=latency_ms,
            accessed_chunks=titles,
            accessed_generated_questions=questions,
            satisfied=None,
        )
        db.add(log)

        # Remember this turn for the next question. "No result" answers are
        # skipped so they don't skew what a follow-up is taken to refer to.
        if result.source != "fallback":
            save_turns(db, org_id, body.session_id, sanitized_query, result.answer)

        await db.flush()
        new_id = log.id
        await db.commit()
        result.response_id = new_id
    except Exception as e:
        await db.rollback()
        logger.exception("Failed to record response for org %s", org_id)

        raise HTTPException(
            status_code=500,
            detail="Failed to save response log; feedback is unavailable.\n error: " + str(e),
        )

    # Housekeeping: delete messages past their expiry (throttled, never raises)
    await purge_expired(db)

    return result

RANGE_DAYS = {
    "last_7_days": 7,
    "last_30_days": 30,
    "last_90_days": 90,
    "last_year": 365,
}

# The Philippines is UTC+8 with no daylight saving, so a fixed offset is enough
# and avoids needing the tzdata package on Windows.
LOCAL_OFFSET = timedelta(hours=8)


def granularity_for(range_key: str) -> str:
    """Bucket size for the chart: keeps the number of points readable."""
    if range_key == "all_time":
        return "month"
    if range_key == "last_year":
        return "week"
    return "day"


def bucket_floor(dt: datetime, granularity: str) -> datetime:
    dt = dt.replace(hour=0, minute=0, second=0, microsecond=0)
    if granularity == "week":
        return dt - timedelta(days=dt.weekday())  # weeks start on Monday
    if granularity == "month":
        return dt.replace(day=1)
    return dt


def bucket_next(dt: datetime, granularity: str) -> datetime:
    if granularity == "week":
        return dt + timedelta(days=7)
    if granularity == "month":
        if dt.month == 12:
            return dt.replace(year=dt.year + 1, month=1)
        return dt.replace(month=dt.month + 1)
    return dt + timedelta(days=1)


def range_start(range_key: str) -> datetime | None:
    days = RANGE_DAYS.get(range_key)
    if days is None:  # "all_time"
        return None
    # naive UTC, to match datetime.utcnow() in the model
    return datetime.now(timezone.utc).replace(tzinfo=None) - timedelta(days=days)


@router.post("", response_model=StatsResponse)  # final URL: POST /stats
async def get_stats(
    body: StatsRequest,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    # --- access control ---
    org = await db.scalar(select(Org).where(Org.id == body.org_id))
    if not org:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND, detail="Organization not found"
        )

    org_member = await db.scalar(
        select(OrgMember).where(
            OrgMember.org_id == body.org_id,
            OrgMember.user_id == current_user.id,
        )
    )
    # Deny only if the user is neither the creator nor a member
    if org.creator_id != current_user.id and not org_member:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="You are not a member of this organization",
        )

    # --- shared filter for the selected range ---
    conditions = [Response.org_id == body.org_id]
    start = range_start(body.range)
    if start is not None:
        conditions.append(Response.created_at >= start)

    # --- aggregates, computed by the database ---
    inquiry_count = (
        await db.scalar(select(func.count()).select_from(Response).where(*conditions))
        or 0
    )

    average_response_time = float(
        await db.scalar(select(func.avg(Response.latency_ms)).where(*conditions))
        or 0.0
    )

    # Only rated answers count: satisfied is NULL until someone rates it
    rated = (
        await db.scalar(
            select(func.count())
            .select_from(Response)
            .where(*conditions, Response.satisfied.is_not(None))
        )
        or 0
    )
    liked = (
        await db.scalar(
            select(func.count())
            .select_from(Response)
            .where(*conditions, Response.satisfied.is_(True))
        )
        or 0
    )
    satisfaction_rate = round(liked / rated * 100, 1) if rated else 0.0  # percent

    # --- top-10 lists, counted in Python from only the columns we need ---
    chunk_rows = await db.execute(
        select(Response.accessed_chunks, Response.created_at).where(*conditions)
    )
    chunk_counts: Counter[str] = Counter()
    chunk_last_accessed: dict[str, datetime] = {}
    for titles, created_at in chunk_rows:
        for title in titles or []:
            chunk_counts[title] += 1
            if title not in chunk_last_accessed or created_at > chunk_last_accessed[title]:
                chunk_last_accessed[title] = created_at

    # Needs a `query` text column on Response; returns {} until you add it
    inquiry_counts: Counter[str] = Counter()
    query_col = getattr(Response, "query", None)
    if query_col is not None:
        query_rows = await db.scalars(select(query_col).where(*conditions))
        for text in query_rows:
            normalized = (text or "").strip()
            if normalized:
                inquiry_counts[normalized.lower()] += 1

    top_chunks = dict(chunk_counts.most_common(10))

    # --- chart: responses per day/week/month, gaps filled with zeros ---
    granularity = granularity_for(body.range)
    # Inline literals (not bind parameters): Postgres treats the same expression
    # with different parameters as different, which breaks GROUP BY under asyncpg.
    # `granularity` only ever comes from granularity_for(), never from user input.
    bucket = func.date_trunc(
        literal_column(f"'{granularity}'"),
        Response.created_at + literal_column("interval '8 hours'"),  # = LOCAL_OFFSET
    )
    chart_rows = (
        await db.execute(
            select(bucket, func.count())
            .where(*conditions)
            .group_by(literal_column("1"))
            .order_by(literal_column("1"))
        )
    ).all()
    bucket_counts = {row[0].date(): row[1] for row in chart_rows}

    chart_data: list[ChartPoint] = []
    if bucket_counts:
        if start is not None:
            first = bucket_floor(start + LOCAL_OFFSET, granularity)
        else:  # all_time: begin at the first bucket that has data
            first = bucket_floor(
                datetime.combine(min(bucket_counts), datetime.min.time()), granularity
            )
        now_local = datetime.now(timezone.utc).replace(tzinfo=None) + LOCAL_OFFSET
        last = bucket_floor(now_local, granularity)

        current = first
        while current <= last:
            chart_data.append(
                ChartPoint(
                    date=current.date().isoformat(),
                    count=bucket_counts.get(current.date(), 0),
                )
            )
            current = bucket_next(current, granularity)

    return StatsResponse(
        inquiry_count=inquiry_count,
        satisfaction_rate=satisfaction_rate,
        average_response_time=average_response_time,
        top_10_inquiries=dict(inquiry_counts.most_common(10)),
        top_10_chunks=top_chunks,
        chart_data=chart_data,
        top_10_chunks_access_time={
            title: accessed_at.replace(tzinfo=timezone.utc)
            for title, accessed_at in sorted(
                chunk_last_accessed.items(), key=lambda item: item[1], reverse=True
            )[:10]
        },
    )

@router.delete("/{website_id}", status_code=status.HTTP_200_OK)
async def delete_website(
    website_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    """
    Deletes a website, clearing its Qdrant vectors before executing the
    PostgreSQL cascade deletion.
    """
    stmt = (
        select(Website)
        .join(OrgMember, Website.org_id == OrgMember.org_id)
        .where(
            Website.id == website_id,
            OrgMember.user_id == current_user.id,
        )
    )
    res = await db.execute(stmt)
    website = res.scalar_one_or_none()

    if not website:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Website not found or unauthorized.",
        )
    await require_org_role(
        db,
        website.org_id,
        current_user,
        CONTENT_MANAGER_ROLES,
        "Only organization administrators and ingesters can manage websites.",
    )

    # Clean up Qdrant points across all pages for this website
    vector_db = get_vector_db()
    try:
        await vector_db.delete_points(
            collection_name=settings.resolved_collection_name,
            filters={"group_id": str(website.org_id), "web_id": website.id},
        )
    except Exception as e:
        # Proceed with deletion if collection or points do not exist
        pass
    finally:
        await vector_db.close()

    # PostgreSQL CASCADE cleans up pages, schedules, chunks, and questions
    await db.delete(website)
    await db.commit()

    return {"message": f"Website {website_id} and its vector indices were deleted."}

def _build_progress_bar(current: int, total: int, length: int = 10) -> str:
    if total <= 0:
        return "░" * length + " (0%)"
    pct = min(max(current / total, 0.0), 1.0)
    filled = int(round(length * pct))
    bar = "█" * filled + "░" * (length - filled)
    return f"{bar} ({int(pct * 100)}%)"

async def start_telegram_bot_listener():
    if not TELEGRAM_BOT_TOKEN:
        print("[Telegram] TELEGRAM_BOT_TOKEN not configured. Skipping listener.")
        return

    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/getUpdates"
    offset = None
    print("[Telegram] Poller active. Listening for commands...")

    async with httpx.AsyncClient(timeout=30.0) as client:
        while True:
            try:
                params = {"timeout": 20}
                if offset:
                    params["offset"] = offset

                resp = await client.get(url, params=params)
                if resp.status_code != 200:
                    await asyncio.sleep(5)
                    continue

                data = resp.json()
                for update_obj in data.get("result", []):
                    offset = update_obj["update_id"] + 1
                    message = update_obj.get("message", {})
                    chat_id = message.get("chat", {}).get("id")
                    text = message.get("text", "").strip()

                    if not chat_id or not text:
                        continue

                    cmd_parts = text.split()
                    command = cmd_parts[0].split("@")[0].lower()

                    if command == "/scrape":
                        if len(cmd_parts) < 2 or not cmd_parts[1].isdigit():
                            await send_reply(
                                chat_id,
                                "⚠️ *Invalid Format!*\n"
                                "Usage: `/scrape <website_id> [force]`\n"
                                "Examples:\n"
                                "• `/scrape 1` (Standard refresh check)\n"
                                "• `/scrape 1 force` (Force re-crawl & re-chunk all pages)",
                            )
                            continue

                        web_id = int(cmd_parts[1])
                        force = len(cmd_parts) >= 3 and cmd_parts[2].lower() in ("force", "true", "yes", "1")
                        asyncio.create_task(
                            run_pipeline_with_notifications(website_id=web_id, chat_id=chat_id, force_refresh=force)
                        )

                    elif command == "/status":

"""Academic-period and temporal helpers shared by ingestion and query time."""
import re
from collections import Counter
from datetime import date, datetime
from typing import Optional, Tuple
from urllib.parse import urlparse

# Standard academic periods: "AY 2025-2026", "A.Y. 2025-26", "SY2025/2026", "academic year 2025-2026"
_AY_RE = re.compile(
    r"(?:\ba\.?\s?y\.?|\bs\.?\s?y\.?|academic\s+year|school\s+year)\s*[:\-]?\s*"
    r"((?:19|20)\d{2})\s*[-\u2013\u2014/]\s*((?:19|20)?\d{2})(?!\d)",
    re.IGNORECASE,
)

# Term/Semester anchors: "1st Sem 2025-2026", "Term 2 2026"
_SEM_RE = re.compile(
    r"\b(?:1st|2nd|3rd|first|second|third|midyear|summer)?\s*(?:sem(?:ester)?|term|trimester)\s*"
    r"[:\-]?\s*(?:ay|sy)?\s*((?:19|20)\d{2})",
    re.IGNORECASE,
)

# Standard four-digit consecutive year ranges: "2025-2026"
_RANGE_RE = re.compile(r"(?<!\d)((?:19|20)\d{2})\s*[-\u2013\u2014/]\s*((?:19|20)\d{2})(?!\d)")

# ISO Dates: "2026-08-15" or "2026/08/15"
_ISO_DATE_RE = re.compile(r"(?<!\d)((?:19|20)\d{2})[-/](?:0[1-9]|1[0-2])[-/](?:0[1-9]|[12]\d|3[01])(?!\d)")

# Bare Year: "2025" or "2026"
_YEAR_RE = re.compile(r"(?<!\d)((?:19|20)\d{2})(?!\d)")


def _end_year(start: int, raw: str) -> int:
    end = int(raw)
    if end < 100:
        end += (start // 100) * 100
    return end


def extract_doc_period(
    text: str,
    *,
    url: str = "",
    heading: str = "",
    allow_bare_year: bool = False,
) -> Optional[int]:
    """
    Returns the target starting year of the academic or institutional period.
    Captures academic years, semesters, ISO document stamps, and URL structures.
    """
    hay = f"{heading}\n{text}"
    counts: Counter = Counter()

    # 1. Direct AY/SY notation (strongest signal: weight 4)
    for m in _AY_RE.finditer(hay):
        start = int(m.group(1))
        if _end_year(start, m.group(2)) == start + 1:
            counts[start] += 4

    # 2. Semester with year reference (weight 3)
    for m in _SEM_RE.finditer(hay):
        start = int(m.group(1))
        counts[start] += 3

    # 3. ISO format timestamps (weight 3)
    for m in _ISO_DATE_RE.finditer(hay):
        start = int(m.group(1))
        counts[start] += 3

    # 4. Standard year range "2025-2026" (weight 2)
    for m in _RANGE_RE.finditer(hay):
        start = int(m.group(1))
        if int(m.group(2)) == start + 1:
            counts[start] += 2

    # 5. URL path patterns: /2025/ or /ay-2025-2026/ (weight 2)
    path = urlparse(url).path if url else ""
    for m in _YEAR_RE.finditer(path):
        counts[int(m.group(1))] += 2

    # 6. Bare year extraction for user queries (e.g., "admissions 2026")
    if allow_bare_year:
        for m in _YEAR_RE.finditer(hay):
            counts[int(m.group(1))] += 1

    if not counts:
        return None

    top_count = max(counts.values())
    return max(year for year, count in counts.items() if count == top_count)


def current_academic_year_start(today: Optional[date] = None) -> int:
    """Calculates active academic anchor safely without regional month bias."""
    today = today or date.today()
    # If past Q1, current year is typically the active starting academic cycle
    return today.year if today.month >= 5 else today.year - 1


def format_period(start_year: Optional[int]) -> Optional[str]:
    return f"{start_year}-{start_year + 1}" if isinstance(start_year, int) else None

import uuid
from typing import Optional

KB_NAMESPACE: uuid.UUID = uuid.UUID("a3bb189e-8bf9-4888-9912-ace4e6543002")


def generate_uuid(*, seed: str) -> str:
    """Generates a deterministic UUIDv5 within the KB namespace."""
    return str(uuid.uuid5(KB_NAMESPACE, seed))

def generate_doc_id(
    *,
    tenant_id: str,
    source_url: str,
) -> str:
    """Generates a deterministic document-level ID based on source URL."""
    seed = f"doc::{tenant_id}::{source_url.strip()}"
    return generate_uuid(seed=seed)


def generate_chunk_id(
    *,
    tenant_id: str,
    source_url: str,
    chunk_index: int,
    content: str,
) -> str:
    """Combines document origin, sequential index, and text to prevent collisions."""
    seed = f"chunk::{tenant_id}::{source_url}#{chunk_index}:{content.strip()}"
    return generate_uuid(seed=seed)


def generate_question_id(
    *,
    chunk_id: str,
    question: str,
) -> str:
    """Generates a deterministic ID bound to the parent chunk and question text."""
    seed = f"question::{chunk_id}:{question.strip()}"
    return generate_uuid(seed=seed)


def generate_sparse_id(
    *,
    parent_id: str,
    model_name: Optional[str] = "bm25",
) -> str:
    """Generates a deterministic ID for a sparse representation tied to its source entity."""
    seed = f"sparse::{model_name}:{parent_id}"
    return generate_uuid(seed=seed)

    # 1. Database schema migrations / creation
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)

    # 2. Vector DB bootstrap: guarantees collection and group_id payload index exist

    poller_task.cancel()
    try:
        await poller_task
    except asyncio.CancelledError:
        pass

    await vector_db.close()
